// Block data made of the token values 0xFF, 0xFE, 0xFD, 0x02 and 0x01.
// They are data, since count says where the run ends.

// header
0x64, 0x6F, 0x74, 0x73,   // fourcc "dots"
0x00, 0x01, 0x00,         // dots-version: major 0, minor 1, update 0
0x00,                     // flags: bit 0 clear (no contains-text)
0x10, 0x04,               // dimensions: 16 columns, 4 rows
0x0A,                     // frame-rate: 10 Hz

// frame 1, block layer
0x02, 0x00, 0x00, 0x05,   // run: position (0,0), count 5
0xFF, 0xFE, 0xFD,         //   data: 3 blocks, not tokens
0x02, 0x01,               //   data: 2 blocks, not tokens
0xFE,                     // end-of-blocks

// frame 1, text layer
0xFD,                     // end-of-text, no runs

// end
0xFF                      // end-of-stream
