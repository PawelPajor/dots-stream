// DOTS 0.1.x. Accept: Two block runs of one frame cover cell (1,0); the
// later one wins, leaving 0xFF 0x00 0xFF.

// header
0x64, 0x6F, 0x74, 0x73,   // fourcc "dots"
0x00, 0x01, 0x00,         // dots-version: major 0, minor 1, update 0
0x00,                     // flags: bit 0 clear (no contains-text)
0x10, 0x04,               // dimensions: 16 columns, 4 rows
0x0A,                     // frame-rate: 10 Hz

// frame 1, block layer
0x02, 0x00, 0x00, 0x03,   // run: position (0,0), count 3
0xFF, 0xFF, 0xFF,         //   data: 3 blocks
0x02, 0x01, 0x00, 0x01,   // run: position (1,0), count 1
0x00,                     //   data: 1 block, 0x00, overwrites (1,0)
0xFE,                     // end-of-blocks

// frame 1, text layer
0xFD,                     // end-of-text, no runs

// end
0xFF                      // end-of-stream
