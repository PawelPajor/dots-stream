// DOTS 0.1.x. Reject: INVALID_VALUE at offset 8. A grid with 0 columns.
// The stream has no frames, so that no run is also outside the grid.

// header
0x64, 0x6F, 0x74, 0x73,   // fourcc "dots"
0x00, 0x01, 0x00,         // dots-version: major 0, minor 1, update 0
0x00,                     // flags: bit 0 clear (no contains-text)
0x00, 0x04,               // dimensions: 0 columns, 4 rows
0x0A,                     // frame-rate: 10 Hz

// end
0xFF                      // end-of-stream
